# Task 29 — BGA Via-in-Pad Batch Plan
# -----------------------------------
# Input files
#   devices.json        list of devices with refdes, pin_count, pitch_mm,
#                       part, and a pins list.  Each pin has
#                       {pin, net, type}; type ∈ {signal, power, ground}.
#   base_padstack.txt   baseline padstack "vh_0p2" definition
#                       (drill 0.20 mm, pad 0.45 mm, plated).
#
# Rule for device eligibility
# ----------------------------
# A device is eligible for via-in-pad conversion iff ALL of:
#   1. Its refdes matches the regex ^U\d+$ (letter U followed by one or
#      more digits, nothing else).
#   2. pin_count >= 64.
#   3. pitch_mm  <= 0.8 (inclusive).
#
# For each eligible device, ONLY its pins whose `type` equals "signal"
# (case-sensitive) receive the via-in-pad treatment.  Power and ground
# pins keep their original padstack (the plan does NOT list them).
#
# Plan output: plan.json in the run's output directory.
#
# Schema
# ------
# {
#   "devices_modified": [<refdes>, ... sorted by ASCII asc],
#   "padstack_ops": [
#     {
#       "name":         "vh_0p2_<refdes>",
#       "derived_from": "vh_0p2",
#       "drill_mm":     0.20,
#       "pad_round_mm": 0.45,
#       "plated":       "Yes",
#       "applied_refdes": "<refdes>"
#     },
#     ... one per eligible device, in the same order as devices_modified
#   ],
#   "pin_ops": [
#     {
#       "refdes":   "<refdes>",
#       "pin":      "<pin-name e.g. A1>",
#       "net":      "<net>",
#       "padstack": "vh_0p2_<refdes>"
#     },
#     ... sorted primarily by refdes ASC, secondarily by pin ASC
#         (ASCII compare: so A1 < A10 < A2 < B1 etc.)
#   ],
#   "summary": {
#     "devices_eligible": <int>,
#     "padstacks_created": <int>,
#     "pins_converted":   <int>
#   }
# }
#
# Numeric fields in the padstack_ops entries are floats with the same
# value as base_padstack.txt (no re-rounding).  `plated` is always the
# literal string "Yes".  Padstack naming: append `_` + refdes to the
# base padstack name.
